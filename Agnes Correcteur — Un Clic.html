function isValidKey(k) { return typeof k === 'string' && k.startsWith('sk-') && k.length >= 15; }

async function quickPing(key) {
  try {
    const res = await fetch(API_BASE + '/chat/completions', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ model: MODEL_TEXT, messages: [{ role: 'user', content: 'ok' }], max_tokens: 1 })
    });
    return res.status !== 401 && res.status !== 403;
  } catch (e) { return null; }
}

async function validateAndSaveKeys() {
  const raw = $('keysInput').value.split('\n').map(l => l.trim()).filter(l => l.length > 0).filter(l => isValidKey(l));
  if (raw.length === 0) return toast('Aucune clé valide format', 'error');
  $('validateBtn').disabled = true;
  $('validateBtn').textContent = '⚡ Validation parallèle…';
  human(`⚡ Validation de ${raw.length} clés en parallèle…`);
  debug(`${raw.length} clés à valider`, 'api');

  const results = await Promise.allSettled(raw.map(k => quickPing(k)));
  state.apiKeys = [];
  state.keyHealth = [];
  state.keyScores = [];

  results.forEach((r, i) => {
    if (r.status === 'fulfilled' && r.value) {
      state.apiKeys.push(raw[i]);
      state.keyHealth.push({ dead: false, lastSuccess: Date.now(), last429: 0, consecutiveErrors: 0 });
      state.keyScores.push(100);
    } else if (r.status === 'fulfilled' && r.value === false) {
      debug(`Clé ${i+1} rejetée (auth)`, 'err');
    }
  });

  $('validateBtn').disabled = false;
  $('validateBtn').textContent = '⚡ Valider (parallèle) et enregistrer';
  $('keyBadge').textContent = `${state.apiKeys.length} clés`;
  $('keyStatus').className = state.apiKeys.length > 0 ? 'status ok' : 'status err';
  $('keyStatusText').textContent = state.apiKeys.length > 0 ? `${state.apiKeys.length} clé(s) opérationnelle(s)` : 'Aucune clé valide';
  renderKeysPanel();
  updateUI();
  human(`✅ ${state.apiKeys.length} clés prêtes.`);
}

function renderKeysPanel() {
  const panel = $('keysPanel');
  panel.innerHTML = state.apiKeys.map((k, i) => {
    const h = state.keyHealth[i];
    const cls = h && h.consecutiveErrors >= 3 ? 'dead' : (h && h.last429 > 0 ? 'slow' : '');
    return `<div class="key-chip ${cls}"><span class="dot"></span>key-${i+1} [${k.slice(0,8)}…]</div>`;
  }).join('');
}

function getBestKey() {
  if (state.apiKeys.length === 0) return null;
  let bestIdx = 0, bestScore = -Infinity;
  for (let i = 0; i < state.apiKeys.length; i++) {
    const h = state.keyHealth[i];
    if (h.dead) continue;
    let score = state.keyScores[i] || 100;
    if (h.consecutiveErrors > 0) score -= h.consecutiveErrors * 15;
    if (h.last429 && Date.now() - h.last429 < 60000) score -= 40;
    if (Date.now() - h.lastSuccess < 10000) score += 20;
    if (score > bestScore) { bestScore = score; bestIdx = i; }
  }
  return { key: state.apiKeys[bestIdx], idx: bestIdx };
}

function markKeySuccess(idx) {
  state.keyHealth[idx].consecutiveErrors = 0;
  state.keyHealth[idx].lastSuccess = Date.now();
  state.keyScores[idx] = Math.min(100, (state.keyScores[idx] || 50) + 5);
}

function markKeyFail(idx, is429) {
  state.keyHealth[idx].consecutiveErrors++;
  if (is429) state.keyHealth[idx].last429 = Date.now();
  state.keyScores[idx] = Math.max(0, (state.keyScores[idx] || 50) - 10);
  if (state.keyHealth[idx].consecutiveErrors >= 5) {
    state.keyHealth[idx].dead = true;
    debug(`Clé-${idx+1} marquée morte (${state.keyHealth[idx].consecutiveErrors} erreurs consécutives)`, 'err');
  }
  renderKeysPanel();
}

function compressImg(dataUrl, maxSide) {
  return new Promise(res => {
    const img = new Image();
    img.onload = () => {
      let { width: w, height: h } = img;
      const s = Math.min(1, maxSide / Math.max(w, h));
      w = Math.round(w * s); h = Math.round(h * s);
      const c = document.createElement('canvas');
      c.width = w; c.height = h;
      const x = c.getContext('2d');
      x.fillStyle = '#fff'; x.fillRect(0, 0, w, h);
      x.drawImage(img, 0, 0, w, h);
      res(c.toDataURL('image/jpeg', 0.85));
    };
    img.onerror = () => res(dataUrl);
    img.src = dataUrl;
  });
}

async function handleImageFile(file) {
  if (!file || !file.type.startsWith('image/')) return toast('Fichier invalide', 'error');
  const dataUrl = await new Promise((res) => {
    const r = new FileReader();
    r.onload = e => res(e.target.result);
    r.readAsDataURL(file);
  });
  const compressed = await compressImg(dataUrl, 1024);
  state.images.push(compressed);
  state.imageNames.push(file.name);
  refreshImagesGrid();
  human(`📷 Image ajoutée : ${file.name}`);
}

function refreshImagesGrid() {
  const grid = $('imagesGrid');
  grid.innerHTML = '';
  state.images.forEach((img, i) => {
    const slot = document.createElement('div');
    slot.className = 'slot filled';
    slot.innerHTML = `<button class="rm" data-rm="${i}">✕</button><img src="${img}" alt="">`;
    grid.appendChild(slot);
  });
  const addSlot = document.createElement('div');
  addSlot.className = 'slot';
  addSlot.innerHTML = `<input type="file" accept="image/*" multiple><div class="icon">＋</div><div class="txt"><strong>Ajouter</strong></div>`;
  addSlot.querySelector('input').onchange = (e) => {
    Array.from(e.target.files).forEach(f => handleImageFile(f));
    e.target.value = '';
  };
  grid.appendChild(addSlot);

  grid.querySelectorAll('[data-rm]').forEach(b => {
    b.onclick = (e) => {
      e.stopPropagation();
      const i = parseInt(b.dataset.rm);
      state.images.splice(i, 1);
      state.imageNames.splice(i, 1);
      state.subjects = [];
      $('subjectDescriptions').style.display = 'none';
      refreshImagesGrid();
    };

  $('refBadge').textContent = `${state.images.length} image${state.images.length > 1 ? 's' : ''}`;
  updateUI();
}

async function acquireSlot() {
  while (state.activeRequests >= MAX_CONCURRENT) await sleep(100);
  state.activeRequests++;
}
function releaseSlot() { state.activeRequests = Math.max(0, state.activeRequests - 1); }

// API — img→txt (vision)
// ============================================================
async function apiVision(system, user, imageUrl) {
  await acquireSlot();
  for (let attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
    if (state.stopRequested) { releaseSlot(); throw new Error('Stop'); }
    const bk = getBestKey();
    if (!bk) { releaseSlot(); throw new Error('Aucune clé active'); }
    const key = bk.key;
    const controller = new AbortController();
    const tid = setTimeout(() => controller.abort(), VISION_TIMEOUT);
    try {
      const res = await fetch(API_BASE + '/chat/completions', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: MODEL_TEXT,
          messages: [
            { role: 'system', content: sanitize(system) },
            { role: 'user', content: [{ type: 'text', text: sanitize(user) }, { type: 'image_url', image_url: { url: imageUrl } }] }
          ],
          temperature: 0.2,
          max_tokens: 300
        }),
        signal: controller.signal
      });
      clearTimeout(tid);
      if (res.status === 401 || res.status === 403) { markKeyFail(bk.idx, false); await sleep(500); continue; }
      if (res.status === 429) { markKeyFail(bk.idx, true); await sleep(ADAPTIVE_DELAY * (attempt + 1)); continue; }
      if (!res.ok) { markKeyFail(bk.idx, false); await sleep(2000 * (attempt + 1)); continue; }
      markKeySuccess(bk.idx);
      releaseSlot();
      const d = await res.json();
      const content = d.choices?.[0]?.message?.content;
      if (!content) throw new Error('Réponse vide');
      return content;
    } catch (e) {
      clearTimeout(tid);
      if (e.name === 'AbortError') { markKeyFail(bk.idx, false); await sleep(1000 * (attempt + 1)); continue; }
      if (attempt === MAX_ATTEMPTS - 1) { releaseSlot(); throw e; }
      await sleep(1000 * (attempt + 1));
    }
  }
  releaseSlot();
  throw new Error('Échec vision après ' + MAX_ATTEMPTS + ' tentatives');
}

// API — txt→img (génération, JAMAIS img→img)
// ============================================================
async function apiImageGenerate(prompt) {
  await acquireSlot();
  for (let attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
    if (state.stopRequested) { releaseSlot(); throw new Error('Stop'); }
    const bk = getBestKey();
    if (!bk) { releaseSlot(); throw new Error('Aucune clé active'); }
    const key = bk.key;
    const controller = new AbortController();
    const tid = setTimeout(() => controller.abort(), IMAGE_TIMEOUT);
    try {
      const res = await fetch(API_BASE + '/images/generations', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: MODEL_IMAGE,
          prompt: sanitize(prompt),
          size: IMAGE_SIZE,
          n: 1,
          extra_body: { response_format: 'url' }
        }),
        signal: controller.signal
      });
      clearTimeout(tid);
      if (res.status === 401 || res.status === 403) { markKeyFail(bk.idx, false); await sleep(500); continue; }
      if (res.status === 429 || res.status === 503) { markKeyFail(bk.idx, true); await sleep(Math.min(30000, ADAPTIVE_DELAY * Math.pow(2, attempt))); continue; }
      if (!res.ok) {
        const t = await res.text();
        markKeyFail(bk.idx, false);
        if (attempt === MAX_ATTEMPTS - 1) { releaseSlot(); throw new Error(`HTTP ${res.status} ${t.slice(0, 100)}`); }
        await sleep(3000 * (attempt + 1));
        continue;
      }
      markKeySuccess(bk.idx);
      const d = await res.json();
      const item = d.data?.[0];
      if (item?.url) { releaseSlot(); return item.url; }
      if (item?.b64_json) { releaseSlot(); return `data:image/png;base64,${item.b64_json}`; }
      markKeyFail(bk.idx, false);
      await sleep(2000);
    } catch (e) {
      clearTimeout(tid);
      if (e.name === 'AbortError') { markKeyFail(bk.idx, true); await sleep(5000 * (attempt + 1)); continue; }
      if (attempt === MAX_ATTEMPTS - 1) { releaseSlot(); throw e; }
      await sleep(3000 * (attempt + 1));
    }
  }
  releaseSlot();
  throw new Error('Échec génération après ' + MAX_ATTEMPTS + ' tentatives');
}

function buildFilmPrompts(subjects, totalFrames) {
  const prompts = [];
  const perAct = Math.ceil(totalFrames / 5);

  for (let i = 0; i < totalFrames; i++) {
    const act = Math.floor(i / perAct);
    const actName = ['Ouverture', 'Montée en tension', 'Point de bascule', 'Résolution', 'Dénouement'][act] || `Acte ${act + 1}`;
    const subIdx = i % subjects.length;
    const subject = subjects[subIdx];

    const narrativeBeats = [
      `Close-up establishing shot, ${subject}. The scene breathes with quiet mystery.`,
      `Medium shot, ${subject}. A figure moves through the environment, purpose unknown.`,
      `Wide cinematic framing, ${subject}. The world expands into surreal depth.`,
      `Dynamic composition, ${subject}. Tension builds, shadows lengthen.`,
      `Climactic wide shot, ${subject}. Everything converges, silence before the storm.`,
      `Intimate detail, ${subject}. A single object tells the whole story.`,
      `Panoramic vista, ${subject}. The horizon stretches infinitely, dust and light.`,
      `Reveal shot, ${subject}. What was hidden emerges into pale focus.`,
      `Transition frame, ${subject}. The scene dissolves into pure atmosphere.`,
      `Heroic pose, ${subject}. Standalone, against an endless backdrop.`
    ];

    const beat = narrativeBeats[i % narrativeBeats.length];
    const prompt = `${MOEBIUS_STYLE} Cinematic frame ${i + 1}/${totalFrames}, Act: ${actName}. ${beat}`;
    prompts.push({ prompt, act: actName, subject, frame: i + 1 });
  }
  return prompts;
}

async function analyzeImages() {
  if (state.images.length === 0) return;
  $('analyzeBtn').disabled = true;
  $('subjectDescriptions').style.display = 'none';
  human(`🔍 Extraction texte de ${state.images.length} images…`);
  setStatus('Extraction img→txt en cours…');

  state.subjects = [];
  const sysPrompt = "You are an objective scene describer for a film production. Describe ONLY the physical narrative content: characters (appearance, pose, action), objects, environment. Do NOT mention art style, medium, color palette, or illustration techniques. Describe as if writing a screenplay scene direction. Keep under 40 words per image.";

  const tasks = state.images.map((img, i) =>
    apiVision(sysPrompt, "Extract the narrative content of this frame for a film script. Focus on what is happening, who is present, where they are.", img).then(desc => {
      debug(`Sujet ${i + 1} : ${desc.slice(0, 120)}…`, 'ok');
      return desc.trim();
    }).catch(e => {
      debug(`Erreur extraction ${i + 1} : ${e.message}`, 'err');
      return "A lone figure in a vast alien landscape, distant horizon, silence.";
    })
  );

  const descs = await Promise.allSettled(tasks);
  descs.forEach((d, i) => {
    $('analyzeStatusText').textContent = `Extraction ${i + 1}/${state.images.length}…`;
    state.subjects.push(d.status === 'fulfilled' ? d.value : "A mysterious scene with a solitary figure.");

  $('analyzeStatusText').textContent = `${state.subjects.length} sujets extraits. Modifiables.`;
  $('subjectDescriptions').style.display = 'block';
  human('✅ Extraction terminée.');
  setStatus(null);
  updateUI();
}

async function generateFilm() {
  if (state.subjects.length === 0) return;
  const totalFrames = Math.max(10, Math.min(200, parseInt($('filmLength').value) || 100));

  $('generateBtn').disabled = true;
  $('resultsCard').classList.remove('hidden');
  const grid = $('resultsGrid');
  grid.innerHTML = '';
  const timeline = $('filmTimeline');
  timeline.innerHTML = '';

  const prompts = buildFilmPrompts(state.subjects, totalFrames);
  state.results = prompts.map(p => ({ ...p, status: 'pending', url: null }));

  // Timeline
  state.results.forEach((_, i) => {
    const f = document.createElement('div');
    f.className = 'frame';
    f.id = `tf-${i}`;
    f.textContent = i + 1;
    f.title = prompts[i].prompt.slice(0, 80);
    f.onclick = () => { const el = $(`card-${i}`); if (el) el.scrollIntoView({ behavior: 'smooth', block: 'center' }); };
    timeline.appendChild(f);
  });

  // Grid cards
  state.results.forEach((r, i) => {
    const card = document.createElement('div');
    card.className = 'photo pending';
    card.id = `card-${i}`;
    card.innerHTML = `
      <div class="photo-img"><div class="load"><div class="spin"></div><div class="gen-title">Frame ${i + 1}</div></div></div>
      <div class="photo-meta">
        <div><span class="pbadge gen">txt→img</span><span class="pbadge act">${esc(r.act)}</span></div>
        <div class="ptitle">Frame ${i + 1} / ${totalFrames}</div>
        <div class="pblock"><div class="plabel">Prompt</div><div class="ptext">${esc(r.prompt)}</div></div>
      </div>
    `;
    grid.appendChild(card);
  });

  updateBatchInfo(totalFrames);
  human(`🎬 Génération de ${totalFrames} frames du film…`);
  setStatus(`Génération ${totalFrames} images (txt→img)…`);

  let idx = 0;
  const workers = Array.from({ length: WORKER_COUNT }, () => (async function worker() {
    while (idx < state.results.length) {
      if (state.stopRequested) return;
      const myIdx = idx++;
      const res = state.results[myIdx];
      const card = $(`card-${myIdx}`);
      const tf = $(`tf-${myIdx}`);

      if (card) { card.className = 'photo generating'; card.querySelector('.photo-img').innerHTML = `<div class="load"><div class="spin"></div><div class="gen-title">Frame ${myIdx + 1}…</div></div>`; }
      if (tf) tf.className = 'frame generating';

      try {
        const url = await apiImageGenerate(res.prompt);
        res.url = url;
        res.status = 'done';
        if (card) {
          card.className = 'photo done';
          card.querySelector('.photo-img').innerHTML = `<img src="${url}" loading="lazy">`;
          card.querySelector('.photo-meta').innerHTML += `<a href="${url}" target="_blank" download class="popen">Télécharger</a>`;
        }
        if (tf) tf.className = 'frame done';
        updateProgress();
      } catch (e) {
        res.status = 'error';
        if (card) { card.className = 'photo error'; card.querySelector('.photo-img').innerHTML = `<div class="err">✕ Échec<br><small>${esc(e.message).slice(0, 60)}</small></div>`; }
        if (tf) tf.className = 'frame error';
        debug(`Frame ${myIdx + 1} échec : ${e.message}`, 'err');
        updateProgress();
      }
    }
  })());

  await Promise.all(workers);
  state.stopRequested = false;

  setStatus(null);
  $('generateBtn').disabled = false;
  const done = state.results.filter(r => r.url).length;
  human(`🏁 Film terminé : ${done}/${totalFrames} frames.`);
  toast(`Film généré : ${done}/${totalFrames} frames`, done === totalFrames ? 'success' : 'warn', 4000);
  renderKeysPanel();
}

function updateProgress() {
  const total = state.results.length;
  const done = state.results.filter(r => r.status === 'done').length;
  const failed = state.results.filter(r => r.status === 'error').length;
  const pct = Math.round(((done + failed) / total) * 100);
  const fill = $('progressFill');
  fill.style.width = pct + '%';
  if (pct === 100) fill.classList.add('done');
  $('progressPct').textContent = `${pct}%`;
  $('resultsBadge').textContent = `${done} / ${total}`;
  if (done % 10 === 0 && done > 0) human(`📽️ ${done}/${total} frames…`);
}

function updateBatchInfo(total) {
  $('batchInfo').innerHTML = `
    <span><strong>${total}</strong> frames</span>
    <span><strong>${state.subjects.length}</strong> sujets</span>
    <span><strong>${state.apiKeys.length}</strong> clés actives</span>
    <span><strong>${WORKER_COUNT}</strong> workers</span>
    <span><strong>${MAX_CONCURRENT}</strong> concurrent max</span>
  `;
}
